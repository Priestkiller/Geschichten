cmake_minimum_required(VERSION 3.22)
project(geschichten_gguf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# The distributable APK uses the historical debug signing key. Native CPU
# inference still needs compiler optimization in that build variant.
add_compile_options(-O3)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("../../../../third_party/llama.cpp-d2e54583c7452353eb35d40431281f6ee984332f" llama-build)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("../../../../third_party/sentencepiece-0.2.1" sentencepiece-build EXCLUDE_FROM_ALL)
add_library(geschichten_gguf SHARED gguf-jni.cpp tokenizer-jni.cpp embedding-jni.cpp)
target_include_directories(geschichten_gguf PRIVATE "../../../../third_party/sentencepiece-0.2.1/src")
target_link_libraries(geschichten_gguf PRIVATE llama sentencepiece-static log)
target_link_options(geschichten_gguf PRIVATE "-Wl,-z,max-page-size=16384")
